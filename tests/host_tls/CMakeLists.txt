cmake_minimum_required(VERSION 3.20)
project(defjam_host_tls C)
if(NOT WIN32)
    message(FATAL_ERROR "This fixture checks the native Windows host TLS declaration")
endif()
enable_testing()
add_executable(host_tls test_main.c)
target_include_directories(host_tls PRIVATE ../../src)
if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(host_tls PRIVATE -Werror=unknown-attributes)
endif()
add_test(NAME host_tls_isolation COMMAND host_tls)
set_tests_properties(host_tls_isolation PROPERTIES TIMEOUT 10)
