cmake_minimum_required(VERSION 3.20)
project(defjam_toolchain_paths LANGUAGES C CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "This fixture qualifies native Windows build tools")
endif()
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${REPO}/cmake/windows_resource.cmake")
set(manifest "${REPO}/src/game.manifest")
defjam_windows_resource(resource NAME toolchain_paths VERSION 1.0.0
    DESCRIPTION "Windows toolchain qualification" MANIFEST "${manifest}")
add_library(probe_tls STATIC "${REPO}/tests/host_tls/test_main.c")
target_include_directories(probe_tls PRIVATE "${REPO}/src")
target_compile_definitions(probe_tls PRIVATE main=host_tls_check)
add_executable(toolchain_paths main.cpp "${resource}" "${manifest}")
target_compile_features(toolchain_paths PRIVATE cxx_std_17)
target_link_libraries(toolchain_paths PRIVATE probe_tls d3d11 mfplat mfuuid xaudio2_8 ole32 dbghelp)
if(MINGW AND CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(probe_tls PRIVATE -Werror=unknown-attributes)
    target_compile_options(toolchain_paths PRIVATE -g -gcodeview)
    target_link_options(toolchain_paths PRIVATE
        "-Wl,--pdb=${CMAKE_CURRENT_BINARY_DIR}/toolchain_paths.pdb"
        "-Wl,--Xlink=/map:${CMAKE_CURRENT_BINARY_DIR}/toolchain_paths.map")
endif()
add_test(NAME windows_toolchain_paths COMMAND toolchain_paths)
set_tests_properties(windows_toolchain_paths PROPERTIES TIMEOUT 10)
